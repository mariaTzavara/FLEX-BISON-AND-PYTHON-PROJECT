<program> ::= <statements>

<statements> ::= <statement> | <statements> <statement>

<statement> ::= <create_table> ';' | <select> ';'

<create_table> ::= CREATE TABLE <table_name> '(' <column_list_table> ')' 

<table_name> ::= IDENTIFIER

<column_list_table> ::= <column> | <column_list_table> ',' <column>

<column> ::= <column_name> <type>

<column_name> ::= IDENTIFIER

<type> ::= INT| FLOAT| VARCHAR '(' INT_LITERAL ')'

<select> ::= SELECT <column_list> FROM <table_name> <opt_where> <opt_group_by> <opt_order_by> <opt_limit>

<column_list> ::= '*' | <column_names>

<column_names> ::= IDENTIFIER | <column_names> ',' IDENTIFIER

<opt_where> ::= ε | WHERE <condition>

<opt_group_by> ::= ε | GROUP BY <column_names>

<opt_order_by> ::= ε | ORDER BY <column_names>

<opt_limit> ::= ε | LIMIT INT_LITERAL

<condition> ::= <condition> AND <condition> | <condition> OR <condition> | NOT <condition> | <predicate>

<predicate> ::= IDENTIFIER <operator> <value> | IDENTIFIER IN '(' <value_list> ')' | IDENTIFIER NOT IN '(' <value_list> ')'

<operator> ::= '=' | '!=' | '<' | '>' | '<=' | '>='

<value> ::= INT_LITERAL | FLOAT_LITERAL | STRING_LITERAL

<value_list> ::= <value> | <value_list> ',' <value>
